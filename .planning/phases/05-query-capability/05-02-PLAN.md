async def query(
    self,
    model: str,
    filters: dict[str, Any] | None = None,
    limit: int = 100,
) -> list[dict[str, Any]]:
    """Query records with optional filters."""
    async with self._session_factory() as session:
        stmt = select(DataRecord).where(DataRecord.model_name == model)

        # Apply filters on JSON data column
        if filters:
            for key, value in filters.items():
                # PostgreSQL JSONB extraction: data->>'key' = value
                stmt = stmt.where(DataRecord.data[key].astext == str(value))

        stmt = stmt.limit(limit)

        result = await session.execute(stmt)
        records = result.scalars().all()

        # Return plain dicts - never leak ORM objects
        return [record.data for record in records]
```

    async def query(
        self,
        model: str,
        filters: dict[str, Any] | None = None,
        limit: int = 100,
    ) -> list[dict[str, Any]]:
        """Query records with optional filters.

        Args:
            model: Model name to query
            filters: Optional filter conditions (column: value)
            limit: Maximum records to return (enforced)

        Returns:
            List of matching records as plain dicts
        """
        async with self._session_factory() as session:
            stmt = select(DataRecord).where(DataRecord.model_name == model)

            # Apply filters on JSON data column
            if filters:
                for key, value in filters.items():
                    # SQLite JSON extraction: json_extract(data, '$.key') = value
                    # SQLAlchemy's data[key].astext works for both dialects
                    stmt = stmt.where(DataRecord.data[key].astext == str(value))

            stmt = stmt.limit(limit)

            result = await session.execute(stmt)
            records = result.scalars().all()

            # Return plain dicts - never leak ORM objects to kernel
            return [record.data for record in records]
```

IMPORTANT: The implementation mirrors PostgresDataRepository.query() exactly. SQLAlchemy's JSON column `data[key].astext` generates dialect-appropriate SQL:
- SQLite: json_extract(data, '$.key')
- PostgreSQL: data->>'key'

@pytest.mark.asyncio
async def test_query_returns_all_records_for_model(repository: SQLiteDataRepository) -> None:
    """Query without filters returns all records for model."""
    # Add multiple records
    await repository.add("contacts", "conn1", {"name": "Alice", "status": "active"})
    await repository.add("contacts", "conn2", {"name": "Bob", "status": "active"})
